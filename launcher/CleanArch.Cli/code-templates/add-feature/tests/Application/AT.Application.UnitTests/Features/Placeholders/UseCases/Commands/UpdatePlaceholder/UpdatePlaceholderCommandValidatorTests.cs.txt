using AppTemplate.Application.Features.Placeholders.UseCases.Commands.UpdatePlaceholder;
#if fields
using AppTemplate.Domain.Features.Placeholders.Model.Entities; //?Placeholder
#endif
using Shouldly;
using Xunit;

namespace AppTemplate.Application.UnitTests.Features.Placeholders.UseCases.Commands.UpdatePlaceholder;

public sealed class UpdatePlaceholderCommandValidatorTests
{
    private static readonly UpdatePlaceholderCommandValidator _validator = new();

    private static CancellationToken TestToken => TestContext.Current.CancellationToken;

#if fields
%FValidatorTests%

    private static UpdatePlaceholderCommand AValidCommand() => %FValidCommand%;
#else
    private static UpdatePlaceholderCommand ACommand(string? name, string? description) => new UpdatePlaceholderCommand(Guid.CreateVersion7(), name!, description!);

    [Fact]
    public async Task ANameAndADescription_AreAccepted() =>
        (await _validator.ValidateAsync(ACommand("Groceries", "Milk"), TestToken)).IsValid.ShouldBeTrue();

    [Fact]
    public async Task AnEmptyDescription_IsAccepted() =>
        (await _validator.ValidateAsync(ACommand("Groceries", string.Empty), TestToken)).IsValid.ShouldBeTrue();

    [Theory]
    [InlineData("")]
    [InlineData("   ")]
    [InlineData(null)]
    public async Task ABlankOrMissingName_IsRejected(string? name) =>
        (await _validator.ValidateAsync(ACommand(name, "Milk"), TestToken)).IsValid.ShouldBeFalse();

    [Fact]
    public async Task AMissingDescription_IsRejected() =>
        (await _validator.ValidateAsync(ACommand("Groceries", null), TestToken)).IsValid.ShouldBeFalse();
#endif
}
