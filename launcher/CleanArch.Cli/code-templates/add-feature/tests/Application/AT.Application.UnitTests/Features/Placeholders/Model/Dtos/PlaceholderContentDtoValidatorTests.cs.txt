using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Policies; //?PlaceholderInputPolicy
using Shouldly;
using Xunit;

namespace AppTemplate.Application.UnitTests.Features.Placeholders.Model.Dtos;

public sealed class PlaceholderContentDtoValidatorTests
{
    private static readonly PlaceholderContentDtoValidator _validator = new();

    private static CancellationToken TestToken => TestContext.Current.CancellationToken;

#if fields
%FContentValidatorTests%

    private static PlaceholderContentDto AValidContentDto() => new(
//#each field separator ",\n"
        %field.Sample%
//#end ");"
#else
    [Fact]
    public async Task ANameAndADescription_AreAccepted() =>
        (await _validator.ValidateAsync(new PlaceholderContentDto("Groceries", "Milk"), TestToken)).IsValid.ShouldBeTrue();

    [Fact]
    public async Task AnEmptyDescription_IsAccepted() =>
        (await _validator.ValidateAsync(new PlaceholderContentDto("Groceries", string.Empty), TestToken)).IsValid.ShouldBeTrue();

    [Theory]
    [InlineData("")]
    [InlineData("   ")]
    public async Task ABlankName_IsRejected(string name) =>
        (await _validator.ValidateAsync(new PlaceholderContentDto(name, "Milk"), TestToken)).IsValid.ShouldBeFalse();

    [Fact]
    public async Task ANullName_IsRejected_NotDereferenced() =>
        (await _validator.ValidateAsync(new PlaceholderContentDto(null!, "Milk"), TestToken)).IsValid.ShouldBeFalse();

    [Fact]
    public async Task ANullDescription_IsRejected() =>
        (await _validator.ValidateAsync(new PlaceholderContentDto("Groceries", null!), TestToken)).IsValid.ShouldBeFalse();

    [Fact]
    public async Task ANameMeasuredAfterTrimming_MayFillTheLimitExactly()
    {
        string name = new('a', %Limits%.NameMaxLength);

        (await _validator.ValidateAsync(new PlaceholderContentDto($"  {name}  ", "Milk"), TestToken)).IsValid.ShouldBeTrue();
        (await _validator.ValidateAsync(new PlaceholderContentDto(name + "a", "Milk"), TestToken)).IsValid.ShouldBeFalse();
    }
#endif
}
