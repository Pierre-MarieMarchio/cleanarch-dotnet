using AppTemplate.Application.Features.Placeholders.Model.Policies; //?PlaceholderInputPolicy|PlaceholderCollectionPolicy
#if agg
using AppTemplate.Domain.Features.Placeholders.Model.Entities; //?Placeholder
#endif
using AppTemplate.Infrastructure.Core.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata;
using Shouldly;
using Xunit;

namespace AppTemplate.Infrastructure.Persistence.UnitTests.Features.Placeholders.Rows.Configurations;

public sealed class PlaceholderRecordConfigurationTests
{
#if etag
    [Fact]
    public void TheVersion_IsTheDatabasesOwnConcurrencyToken()
    {
        var version = Entity().FindProperty(nameof(PlaceholderRecord.Version))!;

        version.IsConcurrencyToken.ShouldBeTrue();
        version.GetColumnName().ShouldBe(DatabaseProvider.VersionColumnName);
    }
#else
    [Fact]
    public void NoPropertyIsAConcurrencyToken() =>
        Entity().GetProperties().ShouldAllBe(property => !property.IsConcurrencyToken);
#endif

#if fields
%FConfigurationTests%
#else
    [Theory]
    [InlineData(nameof(PlaceholderRecord.Name), %Limits%.NameMaxLength)]
    [InlineData(nameof(PlaceholderRecord.Description), %Limits%.DescriptionMaxLength)]
    public void EveryTextColumn_IsAsLongAsTheValidatorAllows(string property, int expected) =>
        Entity().FindProperty(property)!.GetMaxLength().ShouldBe(expected);
#endif

#if list
    [Fact]
    public void EverySortIndex_ServesASortableField_AndEndsWithTheId()
    {
        var indexes = Entity().GetIndexes().ToList();

        indexes.Count.ShouldBe(PlaceholderCollectionPolicy.Instance.SortableFields.Count);

        foreach (var index in indexes)
        {
            var names = index.Properties.Select(property => property.Name).ToList();

#if owner
            names[0].ShouldBe(nameof(PlaceholderRecord.OwnerId));
#endif
            names[^1].ShouldBe(nameof(PlaceholderRecord.Id));
        }
    }

#endif
    [Fact]
#if prefixed
    public void TheTable_IsPrefixedWithThePlaceholdersSchema()
    {
        Entity().GetTableName().ShouldBe($"{AppDbContext.PlaceholdersSchema}_Placeholders");
        Entity().GetSchema().ShouldBeNull();
    }
#else
    public void TheTable_LivesInItsOwnSchema()
    {
        Entity().GetTableName().ShouldBe("Placeholders");
        Entity().GetSchema().ShouldBe(AppDbContext.PlaceholdersSchema);
    }
#endif

    private static IEntityType Entity()
    {
        using var context = new AppDbContext(new DbContextOptionsBuilder<AppDbContext>()
            .UseDatabase(DatabaseProvider.DesignTimeConnectionString)
            .Options);

        return context.Model.FindEntityType(typeof(PlaceholderRecord))!;
    }
}
