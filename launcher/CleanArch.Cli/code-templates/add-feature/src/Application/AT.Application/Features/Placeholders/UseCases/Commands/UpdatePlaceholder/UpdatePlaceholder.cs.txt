using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Core.Common.Validation;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Mapping;
using AppTemplate.Application.Features.Placeholders.Services;
using Mediator;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Commands.UpdatePlaceholder;

public sealed record UpdatePlaceholderCommand(
    Guid PlaceholderId,
//#each field separator ",\n"
    %field.Type% %field.Name%
//#end ","
    VersionPrecondition? Precondition = null) : ICommand<Result<Versioned<PlaceholderDto>>>;

public sealed class UpdatePlaceholderCommandHandler(
    IPlaceholderService service,
    IUnitOfWork unitOfWork) : ICommandHandler<UpdatePlaceholderCommand, Result<Versioned<PlaceholderDto>>>
{
    public async ValueTask<Result<Versioned<PlaceholderDto>>> Handle(
        UpdatePlaceholderCommand command,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(command);

        var loaded = await service.%Load%(command.PlaceholderId, command.Precondition, cancellationToken);

        if (loaded.IsFailure)
        {
            return loaded.To<Versioned<PlaceholderDto>>();
        }

        var placeholder = loaded.Value;

        var update = DomainGuard.Try(() => placeholder.Update(/*#each field separator ", "*/command.%field.Name%/*#end*/));

        if (update.IsFailure)
        {
            return update.To<Versioned<PlaceholderDto>>();
        }

        await unitOfWork.SaveChangesAsync(cancellationToken);

        return PlaceholderDtoMapping.Detail(placeholder);
    }
}
