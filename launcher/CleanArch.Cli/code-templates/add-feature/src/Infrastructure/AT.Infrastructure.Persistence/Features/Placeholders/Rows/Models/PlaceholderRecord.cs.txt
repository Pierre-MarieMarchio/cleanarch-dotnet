using AppTemplate.Domain.Core.Common.Abstractions; //?IAuditable

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;

#if rec
internal sealed class PlaceholderRecord
#else
internal sealed class PlaceholderRecord : IAuditable
#endif
{
    public Guid Id { get; set; }

#if owner
    public Guid OwnerId { get; set; }

#endif
#if fields
//#each field separator "\n\n"
%field.RecordProperty%
//#end
#else
    public string Name { get; set; } = string.Empty;

    public string Description { get; set; } = string.Empty;
#endif

#if rec
    public DateTimeOffset CreatedAt { get; set; }

    public DateTimeOffset? LastModifiedAt { get; set; }
#if etag

    public uint Version { get; set; }
#endif
#else
    public uint Version { get; set; }

    public DateTimeOffset CreatedAt { get; set; }

    public Guid? CreatedBy { get; set; }

    public DateTimeOffset? LastModifiedAt { get; set; }

    public Guid? LastModifiedBy { get; set; }

    void IAuditable.SetCreated(DateTimeOffset at, Guid? by)
    {
        CreatedAt = at;
        CreatedBy = by;
    }

    void IAuditable.SetLastModified(DateTimeOffset at, Guid? by)
    {
        LastModifiedAt = at;
        LastModifiedBy = by;
    }
#endif
}
