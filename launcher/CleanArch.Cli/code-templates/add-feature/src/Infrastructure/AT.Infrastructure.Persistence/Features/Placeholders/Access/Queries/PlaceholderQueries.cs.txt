using System.Linq.Expressions; //?Expression
using AppTemplate.Application.Core.Common.Collections; //?FeaturePageRequest
using AppTemplate.Application.Core.Common.Concurrency; //?Versioned
using AppTemplate.Application.Core.Common.Results; //?PagedResult
using AppTemplate.Application.Features.Placeholders.Model.Dtos; //?PlaceholderSummaryDto|PlaceholderDto
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderQueries;
using AppTemplate.Domain.Core.Common.Primitives; //?UserId
using AppTemplate.Infrastructure.Persistence.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models; //?PlaceholderRecord
using Microsoft.EntityFrameworkCore;

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Access.Queries;

internal sealed class PlaceholderQueries(AppDbContext context) : IPlaceholderQueries
{
#if list
    public async Task<PagedResult<PlaceholderSummaryDto>> %ListQuery%(
#if owner
        UserId ownerId,
#endif
        FeaturePageRequest<PlaceholderFilter> request,
        CancellationToken cancellationToken = default)
    {
#if owner
        ArgumentNullException.ThrowIfNull(ownerId);
#endif
        ArgumentNullException.ThrowIfNull(request);

        IQueryable<PlaceholderRecord> scoped = context.Placeholders.AsNoTracking();
#if owner
        scoped = scoped.Where(record => record.OwnerId == ownerId.Value);
#endif

        return request.Paging.Mode == PagingMode.Offset
            ? await GetOffsetPageAsync(scoped, request, cancellationToken)
            : await GetKeysetPageAsync(scoped, request, cancellationToken);
    }

#endif
#if single
    public Task<%ReadDto%?> GetAsync(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
        CancellationToken cancellationToken = default)
    {
#if owner
        ArgumentNullException.ThrowIfNull(ownerId);

#endif
        return context.Placeholders
            .AsNoTracking()
            .Where(record => record.Id == placeholderId%OwnerClause%)
#if etag
            .Select(record => new Versioned<PlaceholderDto>(
#if fields
                new PlaceholderDto(record.Id, /*#each field separator ", "*/record.%field.Name%/*#end*/, record.CreatedAt, record.LastModifiedAt),
#else
                new PlaceholderDto(record.Id, record.Name, record.Description, record.CreatedAt, record.LastModifiedAt),
#endif
                record.Version))
#else
            .Select(record => new PlaceholderDto(
                record.Id,
#if fields
//#each field
                record.%field.Name%,
//#end
#else
                record.Name,
                record.Description,
#endif
                record.CreatedAt,
                record.LastModifiedAt))
#endif
            .FirstOrDefaultAsync(cancellationToken);
    }

#endif
#if owner
    public async Task<IReadOnlyList<UserId>> ListDataOwnerIdsAsync(
        CancellationToken cancellationToken = default)
    {
        var ids = await context.Placeholders
            .AsNoTracking()
            .Select(record => record.OwnerId)
            .Distinct()
            .ToListAsync(cancellationToken);

        return [.. ids.Select(UserId.Create)];
    }

#endif
#if list
    private static async Task<PagedResult<PlaceholderSummaryDto>> GetOffsetPageAsync(
        IQueryable<PlaceholderRecord> scoped,
        FeaturePageRequest<PlaceholderFilter> request,
        CancellationToken cancellationToken)
    {
        int totalCount = await scoped.CountAsync(cancellationToken);

        int page = request.Paging.Page!.Value;
        int pageSize = request.Paging.PageSize;

        var items = await PlaceholderSortMap.ApplyOrder(scoped, request.Sort)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .Select(_toSummary)
            .ToListAsync(cancellationToken);

        return PagedResult.Offset(items, page, pageSize, totalCount);
    }

    private static async Task<PagedResult<PlaceholderSummaryDto>> GetKeysetPageAsync(
        IQueryable<PlaceholderRecord> scoped,
        FeaturePageRequest<PlaceholderFilter> request,
        CancellationToken cancellationToken)
    {
        var term = request.Sort.Terms[0];
        int pageSize = request.Paging.PageSize;

        var keysetSource = request.Paging.Cursor is { } cursor
            ? PlaceholderSortMap.ApplyKeyset(scoped, term, cursor)
            : scoped;

        var items = await PlaceholderSortMap.ApplyOrder(keysetSource, request.Sort)
            .Take(pageSize + 1)
            .Select(_toSummary)
            .ToListAsync(cancellationToken);

        bool hasNext = items.Count > pageSize;
        var page = hasNext ? items.GetRange(0, pageSize) : items;

        string? nextCursor = null;

        if (hasNext)
        {
            var last = page[^1];

            nextCursor = Cursor.After(term, PlaceholderSortMap.KeyOf(last, term.Field), last.Id).Encode();
        }

        return PagedResult.Keyset(page, pageSize, nextCursor);
    }

    private static readonly Expression<Func<PlaceholderRecord, PlaceholderSummaryDto>> _toSummary =
#if fields
        record => new PlaceholderSummaryDto(record.Id, /*#each field where summary*/record.%field.Name%, /*#end*/record.CreatedAt, record.LastModifiedAt);
#else
        record => new PlaceholderSummaryDto(record.Id, record.Name, record.CreatedAt, record.LastModifiedAt);
#endif
#endif
}
