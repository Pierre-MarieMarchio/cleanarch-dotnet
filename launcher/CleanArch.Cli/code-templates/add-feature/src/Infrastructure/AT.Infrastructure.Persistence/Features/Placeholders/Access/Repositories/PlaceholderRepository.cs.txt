using AppTemplate.Domain.Core.Common.Primitives; //?UserId
using AppTemplate.Domain.Features.Placeholders.Model.Entities;
using AppTemplate.Domain.Features.Placeholders.Repositories;
using AppTemplate.Infrastructure.Persistence.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Mapping;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models; //?PlaceholderRecord
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Tracking;
using Microsoft.EntityFrameworkCore; //?FirstOrDefaultAsync|ToListAsync

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Access.Repositories;

internal sealed class PlaceholderRepository(
    AppDbContext context,
    IPlaceholderMapper mapper,
    IPlaceholderTracker tracker) : IPlaceholderRepository
{
#if change
#if owner
    public async Task<Placeholder?> GetAsync(UserId ownerId, Guid id, CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(ownerId);

        if (tracker.Find(id) is { } alreadyLoaded)
        {
            return alreadyLoaded.OwnerId == ownerId ? alreadyLoaded : null;
        }

        var record = await context.Placeholders.FirstOrDefaultAsync(
            candidate => candidate.Id == id && candidate.OwnerId == ownerId.Value,
            cancellationToken);

        return record is null ? null : LoadOrTrack(record);
    }
#else
    public async Task<Placeholder?> GetAsync(Guid id, CancellationToken cancellationToken)
    {
        if (tracker.Find(id) is { } alreadyLoaded)
        {
            return alreadyLoaded;
        }

        var record = await context.Placeholders.FirstOrDefaultAsync(
            candidate => candidate.Id == id,
            cancellationToken);

        return record is null ? null : LoadOrTrack(record);
    }
#endif

#endif
#if owner
    public async Task<IReadOnlyList<Placeholder>> GetForOwnerAsync(
        UserId ownerId,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(ownerId);

        var records = await context.Placeholders
            .Where(record => record.OwnerId == ownerId.Value)
            .ToListAsync(cancellationToken);

        var aggregates = new List<Placeholder>(records.Count);

        foreach (var record in records)
        {
            aggregates.Add(LoadOrTrack(record));
        }

        return aggregates;
    }

#endif
#if create
    public void Add(Placeholder aggregate)
    {
        ArgumentNullException.ThrowIfNull(aggregate);

        var record = mapper.ToNewRecord(aggregate);

        context.Placeholders.Add(record);
        tracker.Track(aggregate, record);
    }

#endif
#if (delete || owner)
    public void Remove(Placeholder aggregate)
    {
        ArgumentNullException.ThrowIfNull(aggregate);

        if (tracker.FindRecord(aggregate.Id) is not { } record)
        {
            return;
        }

        context.Placeholders.Remove(record);
        tracker.MarkRemoved(aggregate, record);
    }

#endif
#if (change || owner)
    private Placeholder LoadOrTrack(PlaceholderRecord record)
    {
        var aggregate = mapper.ToAggregate(record);

        tracker.Track(aggregate, record);

        return aggregate;
    }
#endif
}
