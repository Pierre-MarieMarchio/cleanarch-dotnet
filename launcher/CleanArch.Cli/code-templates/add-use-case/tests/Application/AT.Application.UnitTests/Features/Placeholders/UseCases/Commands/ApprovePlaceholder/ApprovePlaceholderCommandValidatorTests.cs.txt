using AppTemplate.Application.Features.Placeholders.UseCases.Commands.ApprovePlaceholder;
using Shouldly;
using Xunit;

namespace AppTemplate.Application.UnitTests.Features.Placeholders.UseCases.Commands.ApprovePlaceholder;

public sealed class ApprovePlaceholderCommandValidatorTests
{
    private readonly ApprovePlaceholderCommandValidator _validator = new();

    private static CancellationToken TestToken => TestContext.Current.CancellationToken;

    [Fact]
    public async Task AWellFormedCommand_PassesValidation() =>
        (await _validator.ValidateAsync(AWellFormedCommand(), TestToken))
            .IsValid.ShouldBeTrue();

//#each parameter where ruled separator "\n\n"
    [Fact]
    public async Task AnEmpty%parameter.Name%_IsRefused() =>
        (await _validator.ValidateAsync(AWellFormedCommand() with { %parameter.Name% = %parameter.Empty% }, TestToken))
            .IsValid.ShouldBeFalse();
//#end

    internal static ApprovePlaceholderCommand AWellFormedCommand() =>
        new(/*#each parameter separator ", "*/%parameter.Valid%/*#end*/);
}
