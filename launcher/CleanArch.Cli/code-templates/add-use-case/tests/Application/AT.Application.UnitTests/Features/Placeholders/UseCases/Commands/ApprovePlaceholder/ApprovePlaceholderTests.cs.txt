using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.UseCases.Commands.ApprovePlaceholder;
using AppTemplate.Application.UnitTests.TestDoubles;
using Shouldly;
using Xunit;

namespace AppTemplate.Application.UnitTests.Features.Placeholders.UseCases.Commands.ApprovePlaceholder;

public sealed class ApprovePlaceholderTests
{
    private static CancellationToken TestToken => TestContext.Current.CancellationToken;

    [Fact]
    public async Task AMalformedCommand_NeverReachesTheHandler()
    {
//#each parameter where malformed
        var malformed = ApprovePlaceholderCommandValidatorTests.AWellFormedCommand() with { %parameter.Name% = %parameter.Empty% };
//#end

        var result = await UseCase().Handle(malformed, TestToken);

        result.IsFailure.ShouldBeTrue();
        result.Error!.Type.ShouldBe(ErrorType.Validation);
    }

    private static ValidatedHandler<ApprovePlaceholderCommand, Result> UseCase() =>
        new(new ApprovePlaceholderCommandHandler().Handle, new ApprovePlaceholderCommandValidator());
}
