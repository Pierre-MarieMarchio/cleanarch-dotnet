using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Services;
using AppTemplate.Domain.Features.Placeholders.Repositories;
using Mediator;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Commands.DeletePlaceholder;

public sealed record DeletePlaceholderCommand(Guid PlaceholderId, VersionPrecondition? Precondition = null) : ICommand<Result>;

public sealed class DeletePlaceholderCommandHandler(
    IPlaceholderService service,
    IPlaceholderRepository repository,
    IUnitOfWork unitOfWork) : ICommandHandler<DeletePlaceholderCommand, Result>
{
    public async ValueTask<Result> Handle(
        DeletePlaceholderCommand command,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(command);

        var loaded = await service.%Load%(command.PlaceholderId, command.Precondition, cancellationToken);

        if (loaded.IsFailure)
        {
            return loaded;
        }

        repository.Remove(loaded.Value);
        await unitOfWork.SaveChangesAsync(cancellationToken);

        return Result.Success();
    }
}
