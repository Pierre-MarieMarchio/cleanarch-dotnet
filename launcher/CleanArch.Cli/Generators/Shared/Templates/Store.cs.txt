using AppTemplate.Application.Core.Common.Concurrency; //?VersionPrecondition
using AppTemplate.Application.Features.Placeholders.Model.Dtos; //?PlaceholderContentDto
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderStore;
using AppTemplate.Domain.Core.Common.Primitives; //?UserId
using AppTemplate.Infrastructure.Persistence.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;
using Microsoft.EntityFrameworkCore; //?SingleOrDefaultAsync|ToListAsync

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Access.Stores;

internal sealed class PlaceholderStore(AppDbContext context) : IPlaceholderStore
{
#if create
    public void Add(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
        PlaceholderContentDto content,
        DateTimeOffset createdAt)
    {
#if owner
        ArgumentNullException.ThrowIfNull(ownerId);
#endif
        ArgumentNullException.ThrowIfNull(content);

        context.Placeholders.Add(new PlaceholderRecord
        {
            Id = placeholderId,
#if owner
            OwnerId = ownerId.Value,
#endif
#if fields
%FStoreAssign%
#else
            Name = content.Name,
            Description = content.Description,
#endif
            CreatedAt = createdAt,
        });
    }

#endif
#if update
    public async Task<PlaceholderChangeStatus> StageReplacementAsync(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
        PlaceholderContentDto content,
        DateTimeOffset modifiedAt,
#if etag
        VersionPrecondition? precondition,
#endif
        CancellationToken cancellationToken = default)
    {
#if owner
        ArgumentNullException.ThrowIfNull(ownerId);
#endif
        ArgumentNullException.ThrowIfNull(content);

        var (status, record) = await LoadForChangeAsync(
#if owner
            ownerId,
#endif
            placeholderId,
#if etag
            precondition,
#endif
            cancellationToken);

        if (record is not null)
        {
#if fields
%FStoreUpdate%
#else
            record.Name = content.Name;
            record.Description = content.Description;
#endif
            record.LastModifiedAt = modifiedAt;
        }

        return status;
    }

#endif
#if delete
    public async Task<PlaceholderChangeStatus> StageRemovalAsync(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
#if etag
        VersionPrecondition? precondition,
#endif
        CancellationToken cancellationToken = default)
    {
#if owner
        ArgumentNullException.ThrowIfNull(ownerId);

#endif
        var (status, record) = await LoadForChangeAsync(
#if owner
            ownerId,
#endif
            placeholderId,
#if etag
            precondition,
#endif
            cancellationToken);

        if (record is not null)
        {
            context.Placeholders.Remove(record);
        }

        return status;
    }

#endif
#if change
    private async Task<(PlaceholderChangeStatus Status, PlaceholderRecord? Record)> LoadForChangeAsync(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
#if etag
        VersionPrecondition? precondition,
#endif
        CancellationToken cancellationToken)
    {
        var tracked = await context.Placeholders
            .SingleOrDefaultAsync(record => record.Id == placeholderId%OwnerClause%, cancellationToken);

        if (tracked is null)
        {
            return (PlaceholderChangeStatus.NotFound, null);
        }

#if etag
        return precondition is not null && !precondition.IsSatisfiedBy(tracked.Version)
            ? (PlaceholderChangeStatus.VersionMismatch, null)
            : (PlaceholderChangeStatus.Staged, tracked);
#else
        return (PlaceholderChangeStatus.Staged, tracked);
#endif
    }
#endif
#if purge

    public async Task<int> StageRemovalForOwnerAsync(
        UserId ownerId,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(ownerId);

        var records = await context.Placeholders
            .Where(record => record.OwnerId == ownerId.Value)
            .ToListAsync(cancellationToken);

        context.Placeholders.RemoveRange(records);

        return records.Count;
    }
#endif
#if anon

    public async Task<int> StageAnonymisationForOwnerAsync(
        UserId ownerId,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(ownerId);

        var records = await context.Placeholders
            .Where(record => record.OwnerId == ownerId.Value)
            .ToListAsync(cancellationToken);

        foreach (var record in records)
        {
#if fields
%FStoreAnonymise%
#else
            record.Name = AnonymisedText;
            record.Description = string.Empty;
#endif
        }

        return records.Count;
    }
#if anontext

    private const string AnonymisedText = "[deleted account]";
#endif
#endif
}
