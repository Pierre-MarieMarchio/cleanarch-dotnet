namespace AppTemplate.Application.Features.Placeholders.Services;

#if aggregate
internal sealed class PlaceholderService(IPlaceholderRepository repository, ICurrentUser currentUser) : IPlaceholderService
{
    public async Task<Result<Placeholder>> %Load%(
        Guid placeholderId,
        VersionPrecondition? precondition,
        CancellationToken cancellationToken = default)
    {
        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<Placeholder>();
        }

        var aggregate = await repository.GetAsync(%SvcOwner%placeholderId, cancellationToken);

        if (aggregate is null)
        {
            return Result.Failure<Placeholder>(PlaceholderErrors.PlaceholderNotFound(placeholderId));
        }

        return precondition is not null && !precondition.IsSatisfiedBy(aggregate.Version)
            ? Result.Failure<Placeholder>(ConcurrencyErrors.PreconditionFailed)
            : Result.Success(aggregate);
    }
}
#else
internal sealed class PlaceholderService(
#if reads
    IPlaceholderQueries queries,
#endif
#if writes
    IPlaceholderStore store,
    IUnitOfWork unitOfWork,
#endif
#if stampsTime
    TimeProvider timeProvider,
#endif
#if takesContent
    IValidator<PlaceholderContentDto> validator,
#endif
    ICurrentUser currentUser) : IPlaceholderService
{
#if list
    public async Task<Result<PagedResult<PlaceholderSummaryDto>>> ListAsync(
        PlaceholderListQueryDto query,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(query);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<PagedResult<PlaceholderSummaryDto>>();
        }

        var bound = CollectionBinding.Bind(
            query,
            PlaceholderCollectionPolicy.Instance,
            () => Result.Success(PlaceholderFilter.None),
            %DateKeys%);

        if (bound.IsFailure)
        {
            return bound.To<PagedResult<PlaceholderSummaryDto>>();
        }

        return await queries.%ListQuery%(%SvcOwner%bound.Value, cancellationToken);
    }

#endif
#if get
    public async Task<Result<%ReadDto%>> GetAsync(Guid placeholderId, CancellationToken cancellationToken = default)
    {
        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<%ReadDto%>();
        }

        return await ReadAsync(%SvcOwner%placeholderId, cancellationToken);
    }

#endif
#if create
    public async Task<Result<%ReadDto%>> CreateAsync(
        PlaceholderContentDto content,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(content);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<%ReadDto%>();
        }

        var validation = await validator.EnsureValidAsync(content, cancellationToken);

        if (validation.IsFailure)
        {
            return validation.To<%ReadDto%>();
        }

        var placeholderId = Guid.CreateVersion7();

        store.Add(%SvcOwner%placeholderId, %Staged%, timeProvider.GetUtcNow());
        await unitOfWork.SaveChangesAsync(cancellationToken);

        return await ReadAsync(%SvcOwner%placeholderId, cancellationToken);
    }

#endif
#if update
    public async Task<Result<%ReadDto%>> UpdateAsync(
        Guid placeholderId,
        PlaceholderContentDto content,
#if etag
        VersionPrecondition? precondition,
#endif
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(content);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<%ReadDto%>();
        }

        var validation = await validator.EnsureValidAsync(content, cancellationToken);

        if (validation.IsFailure)
        {
            return validation.To<%ReadDto%>();
        }

        var status = await store.StageReplacementAsync(
#if owner
            userId.Value,
#endif
            placeholderId,
            %Staged%,
            timeProvider.GetUtcNow(),
#if etag
            precondition,
#endif
            cancellationToken);

        if (status != PlaceholderChangeStatus.Staged)
        {
            return Result.Failure<%ReadDto%>(FailureOf(status, placeholderId));
        }

        await unitOfWork.SaveChangesAsync(cancellationToken);

        return await ReadAsync(%SvcOwner%placeholderId, cancellationToken);
    }

#endif
#if delete
    public async Task<Result> DeleteAsync(
        Guid placeholderId,
#if etag
        VersionPrecondition? precondition,
#endif
        CancellationToken cancellationToken = default)
    {
        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId;
        }

        var status = await store.StageRemovalAsync(
#if owner
            userId.Value,
#endif
            placeholderId,
#if etag
            precondition,
#endif
            cancellationToken);

        if (status != PlaceholderChangeStatus.Staged)
        {
            return Result.Failure(FailureOf(status, placeholderId));
        }

        await unitOfWork.SaveChangesAsync(cancellationToken);

        return Result.Success();
    }

#endif
#if readsOne
    private async Task<Result<%ReadDto%>> ReadAsync(
#if owner
        UserId ownerId,
#endif
        Guid placeholderId,
        CancellationToken cancellationToken)
    {
        var placeholder = await queries.GetAsync(%OwnerArg%placeholderId, cancellationToken);

        return placeholder is null
            ? Result.Failure<%ReadDto%>(PlaceholderErrors.PlaceholderNotFound(placeholderId))
            : placeholder;
    }

#endif
#if trimsText
    private static PlaceholderContentDto Trimmed(PlaceholderContentDto content) => content with { /*#each field where string separator ", "*/%field.Name% = %field.Trimmed%/*#end*/ };
#endif
#if changesOne
    private static Error FailureOf(PlaceholderChangeStatus status, Guid placeholderId) => status switch
    {
        PlaceholderChangeStatus.NotFound => PlaceholderErrors.PlaceholderNotFound(placeholderId),
#if etag
        PlaceholderChangeStatus.VersionMismatch => ConcurrencyErrors.PreconditionFailed,
#endif
        _ => throw new ArgumentOutOfRangeException(nameof(status), status, "A staged change has no failure."),
    };
#endif
}
#endif
