using AppTemplate.Application.Core.Common.Collections;
using AppTemplate.Application.Core.Common.Policies;

namespace AppTemplate.Application.Features.Placeholders.Model.Policies;

public sealed class PlaceholderCollectionPolicy : ICollectionPolicy
{
//#each field where sortable separator "\n\n"
    public const string %field.Name%Field = "%field.Parameter%";
//#end

    public const string CreatedAtField = "createdAt";

    public static readonly PlaceholderCollectionPolicy Instance = new();

    public IReadOnlyList<SortableField> SortableFields { get; } =
    [
//#each field where sortable
        SortableField.%field.Sorting%(%field.Name%Field),
//#end
        SortableField.Keyset(CreatedAtField),
    ];

    public string DefaultSort => "createdAt:desc";

    public int MaxSortTerms => 2;

    public int MaxPageSize => 100;

    public int DefaultPageSize => 20;
}
