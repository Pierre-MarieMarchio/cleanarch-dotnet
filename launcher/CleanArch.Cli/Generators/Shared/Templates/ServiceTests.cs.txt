using AppTemplate.Application.Core.Common.Collections; //?FeaturePageRequest
using AppTemplate.Application.Core.Common.Concurrency; //?Versioned|VersionPrecondition
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Model.Dtos; //?PlaceholderContentDto|PlaceholderDto|PlaceholderListQueryDto
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderQueries; //?IPlaceholderQueries|PlaceholderFilter
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderStore; //?IPlaceholderStore|PlaceholderChangeStatus
using AppTemplate.Application.Features.Placeholders.Services;
using AppTemplate.Application.Features.Placeholders.Model.Dtos; //?PlaceholderContentDtoValidator
using AppTemplate.Application.Features.Placeholders.Model.Policies; //?PlaceholderInputPolicy
using AppTemplate.Application.UnitTests.TestDoubles;
using AppTemplate.Domain.Core.Common.Primitives;
using NSubstitute;
using Shouldly;
using Xunit;

namespace AppTemplate.Application.UnitTests.Features.Placeholders.Services;

public sealed class PlaceholderServiceTests
{
    private static readonly UserId _callerId = UserId.Create(Guid.CreateVersion7());

#if reads
    private readonly IPlaceholderQueries _queries = Substitute.For<IPlaceholderQueries>();
#endif
#if writes
    private readonly IPlaceholderStore _store = Substitute.For<IPlaceholderStore>();
    private readonly IUnitOfWork _unitOfWork = Substitute.For<IUnitOfWork>();
#endif

    private static CancellationToken TestToken => TestContext.Current.CancellationToken;

#if takesContent
    private static PlaceholderContentDto AValidContent => new(
//#each field separator ",\n"
        %field.Sample%
//#end ");"

#endif
    [Fact]
    public async Task AnAnonymousCaller_IsRefusedByEveryOperation()
    {
        var service = ServiceFor(StubCurrentUser.Anonymous);
#if (get || update || delete)
        var placeholderId = Guid.CreateVersion7();
#endif

        Result[] results =
        [
#if list
            await service.ListAsync(%ListInput%.Offset(null, null), TestToken),
#endif
#if get
            await service.GetAsync(placeholderId, TestToken),
#endif
#if create
            await service.CreateAsync(AValidContent, TestToken),
#endif
#if update
            await service.UpdateAsync(placeholderId, AValidContent%NullPrecondition%, TestToken),
#endif
#if delete
            await service.DeleteAsync(placeholderId%NullPrecondition%, TestToken),
#endif
        ];

        results.ShouldAllBe(result => result.IsFailure);
        results.ShouldAllBe(result => result.Error!.Type == ErrorType.Unauthorized && result.Error.Code == "auth.required");
    }

    [Fact]
    public async Task AnAnonymousCaller_ReadsAndWritesNothing()
    {
        var service = ServiceFor(StubCurrentUser.Anonymous);
#if (get || update || delete)
        var placeholderId = Guid.CreateVersion7();
#endif

#if list
        await service.ListAsync(%ListInput%.Offset(null, null), TestToken);
#endif
#if get
        await service.GetAsync(placeholderId, TestToken);
#endif
#if create
        await service.CreateAsync(AValidContent, TestToken);
#endif
#if update
        await service.UpdateAsync(placeholderId, AValidContent%NullPrecondition%, TestToken);
#endif
#if delete
        await service.DeleteAsync(placeholderId%NullPrecondition%, TestToken);
#endif

#if reads
        _queries.ReceivedCalls().ShouldBeEmpty();
#endif
#if writes
        _store.ReceivedCalls().ShouldBeEmpty();
        await _unitOfWork.DidNotReceive().SaveChangesAsync(Arg.Any<CancellationToken>());
#endif
    }

#if get
    [Fact]
    public async Task GetAsync_%An%PlaceholderThatIsNotThere_IsNotFound()
    {
        _queries.GetAsync(%AnyOwner%Arg.Any<Guid>(), Arg.Any<CancellationToken>())
            .Returns((%ReadDto%?)null);

        var result = await Service().GetAsync(Guid.CreateVersion7(), TestToken);

        result.Error!.Type.ShouldBe(ErrorType.NotFound);
        result.Error.Code.ShouldBe("placeholder.notFound");
    }

#endif
#if create
%FServiceInvalid%

#endif
#if delete
    [Fact]
    public async Task DeleteAsync_%An%PlaceholderThatCannotBeStaged_FailsAndCommitsNothing()
    {
        _store.StageRemovalAsync(%AnyOwner%Arg.Any<Guid>(), %AnyPrecondition%Arg.Any<CancellationToken>())
            .Returns(PlaceholderChangeStatus.NotFound);

        var result = await Service().DeleteAsync(Guid.CreateVersion7()%NullPrecondition%, TestToken);

        result.Error!.Code.ShouldBe("placeholder.notFound");
        await _unitOfWork.DidNotReceive().SaveChangesAsync(Arg.Any<CancellationToken>());
    }

#endif
    private PlaceholderService ServiceFor(ICurrentUser currentUser) =>
        new(%ServiceArgs%);
#if (get || create || delete)

    private PlaceholderService Service() => ServiceFor(StubCurrentUser.WithId(_callerId));
#endif
}
