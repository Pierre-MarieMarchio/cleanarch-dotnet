using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Core.Common.Validation;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Mapping;
using AppTemplate.Domain.Features.Placeholders.Model.Entities;
using AppTemplate.Domain.Features.Placeholders.Repositories;
using Mediator;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Commands.CreatePlaceholder;

public sealed record CreatePlaceholderCommand(
//#each field separator ",\n"
    %field.Type% %field.Name%
//#end ") : ICommand<Result<Versioned<PlaceholderDto>>>;"

public sealed class CreatePlaceholderCommandHandler(
    IPlaceholderRepository repository,
    IUnitOfWork unitOfWork,
    ICurrentUser currentUser) : ICommandHandler<CreatePlaceholderCommand, Result<Versioned<PlaceholderDto>>>
{
    public async ValueTask<Result<Versioned<PlaceholderDto>>> Handle(
        CreatePlaceholderCommand command,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(command);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<Versioned<PlaceholderDto>>();
        }

        var created = DomainGuard.Try(() => Placeholder.Create(%SvcOwner%/*#each field separator ", "*/command.%field.Name%/*#end*/));

        if (created.IsFailure)
        {
            return created.To<Versioned<PlaceholderDto>>();
        }

        var placeholder = created.Value;

        repository.Add(placeholder);
        await unitOfWork.SaveChangesAsync(cancellationToken);

        return PlaceholderDtoMapping.Detail(placeholder);
    }
}
