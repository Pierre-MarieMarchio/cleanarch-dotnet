using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Domain.Features.Placeholders.Model.Entities;

namespace AppTemplate.Application.Features.Placeholders.Model.Mapping;

internal static class PlaceholderDtoMapping
{
    public static Versioned<PlaceholderDto> Detail(Placeholder placeholder)
    {
        ArgumentNullException.ThrowIfNull(placeholder);

        var detail = new PlaceholderDto(
            placeholder.Id,
#if fields
%FDetailArgs%
#else
            placeholder.Name,
            placeholder.Description,
#endif
            placeholder.CreatedAt,
            placeholder.LastModifiedAt);

        return new Versioned<PlaceholderDto>(detail, placeholder.Version);
    }
}
