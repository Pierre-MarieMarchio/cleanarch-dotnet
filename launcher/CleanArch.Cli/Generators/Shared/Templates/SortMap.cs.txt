using System.Globalization;
using AppTemplate.Application.Core.Common.Collections;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Policies;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Access.Queries;

internal static class PlaceholderSortMap
{
    public static IOrderedQueryable<PlaceholderRecord> ApplyOrder(IQueryable<PlaceholderRecord> source, SortOrder sort)
    {
        ArgumentNullException.ThrowIfNull(source);
        ArgumentNullException.ThrowIfNull(sort);

        IOrderedQueryable<PlaceholderRecord>? ordered = null;

        foreach (var term in sort.Terms)
        {
            ordered = ordered is null ? OrderByTerm(source, term) : ThenByTerm(ordered, term);
        }

        return sort.Terms[^1].Direction == SortDirection.Ascending
            ? ordered!.ThenBy(record => record.Id)
            : ordered!.ThenByDescending(record => record.Id);
    }

    public static IQueryable<PlaceholderRecord> ApplyKeyset(IQueryable<PlaceholderRecord> source, SortTerm term, Cursor cursor)
    {
        ArgumentNullException.ThrowIfNull(source);
        ArgumentNullException.ThrowIfNull(term);
        ArgumentNullException.ThrowIfNull(cursor);

        return term.Field switch
        {
            PlaceholderCollectionPolicy.NameField => ApplyNameKeyset(source, term.Direction, cursor),
            PlaceholderCollectionPolicy.CreatedAtField => ApplyCreatedAtKeyset(source, term.Direction, cursor),
            _ => throw new InvalidOperationException(
                $"'{term.Field}' has no keyset comparison: the policy should have rejected it first."),
        };
    }

    public static string KeyOf(PlaceholderSummaryDto summary, string field)
    {
        ArgumentNullException.ThrowIfNull(summary);

        return field switch
        {
            PlaceholderCollectionPolicy.NameField => summary.Name,
            PlaceholderCollectionPolicy.CreatedAtField => summary.CreatedAt.ToString("O", CultureInfo.InvariantCulture),
            _ => throw new InvalidOperationException(
                $"'{field}' has no cursor key: the policy should have rejected it first."),
        };
    }

    private static IOrderedQueryable<PlaceholderRecord> OrderByTerm(IQueryable<PlaceholderRecord> source, SortTerm term) =>
        term.Field switch
        {
            PlaceholderCollectionPolicy.NameField => term.Direction == SortDirection.Ascending
                ? source.OrderBy(record => record.Name)
                : source.OrderByDescending(record => record.Name),
            PlaceholderCollectionPolicy.CreatedAtField => term.Direction == SortDirection.Ascending
                ? source.OrderBy(record => record.CreatedAt)
                : source.OrderByDescending(record => record.CreatedAt),
            _ => throw new InvalidOperationException(
                $"'{term.Field}' is not a sortable field: the policy should have rejected it first."),
        };

    private static IOrderedQueryable<PlaceholderRecord> ThenByTerm(IOrderedQueryable<PlaceholderRecord> source, SortTerm term) =>
        term.Field switch
        {
            PlaceholderCollectionPolicy.NameField => term.Direction == SortDirection.Ascending
                ? source.ThenBy(record => record.Name)
                : source.ThenByDescending(record => record.Name),
            PlaceholderCollectionPolicy.CreatedAtField => term.Direction == SortDirection.Ascending
                ? source.ThenBy(record => record.CreatedAt)
                : source.ThenByDescending(record => record.CreatedAt),
            _ => throw new InvalidOperationException(
                $"'{term.Field}' is not a sortable field: the policy should have rejected it first."),
        };

    private static IQueryable<PlaceholderRecord> ApplyNameKeyset(
        IQueryable<PlaceholderRecord> source,
        SortDirection direction,
        Cursor cursor) =>
        direction == SortDirection.Ascending
            ? source.Where(record =>
                record.Name.CompareTo(cursor.Key) > 0
                || (record.Name == cursor.Key && record.Id.CompareTo(cursor.Id) > 0))
            : source.Where(record =>
                record.Name.CompareTo(cursor.Key) < 0
                || (record.Name == cursor.Key && record.Id.CompareTo(cursor.Id) < 0));

    private static IQueryable<PlaceholderRecord> ApplyCreatedAtKeyset(
        IQueryable<PlaceholderRecord> source,
        SortDirection direction,
        Cursor cursor)
    {
        if (!DateTimeOffset.TryParse(
            cursor.Key,
            CultureInfo.InvariantCulture,
            DateTimeStyles.RoundtripKind,
            out DateTimeOffset parsed))
        {
            throw new InvalidOperationException(
                "The cursor's key is not a valid date/time: CursorKeys should have rejected it first.");
        }

        var key = parsed.ToUniversalTime();

        return direction == SortDirection.Ascending
            ? source.Where(record =>
                record.CreatedAt.CompareTo(key) > 0
                || (record.CreatedAt == key && record.Id.CompareTo(cursor.Id) > 0))
            : source.Where(record =>
                record.CreatedAt.CompareTo(key) < 0
                || (record.CreatedAt == key && record.Id.CompareTo(cursor.Id) < 0));
    }
}
