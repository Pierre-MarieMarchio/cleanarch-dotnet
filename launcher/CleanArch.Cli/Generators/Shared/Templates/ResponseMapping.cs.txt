using AppTemplate.Api.Common.Contracts; //?PagedResponse
using AppTemplate.Api.Features.Placeholders.Contracts.Responses;
using AppTemplate.Application.Core.Common.Concurrency; //?Versioned
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;

namespace AppTemplate.Api.Features.Placeholders.Contracts.Mapping;

internal static class PlaceholderResponseMapping
{
#if dto
    public static PlaceholderResponse ToResponse(PlaceholderDto placeholder)
    {
        ArgumentNullException.ThrowIfNull(placeholder);

#if fields
        return new PlaceholderResponse(placeholder.Id, %FResponseArgs%placeholder.CreatedAt, placeholder.LastModifiedAt);
#else
        return new PlaceholderResponse(placeholder.Id, placeholder.Name, placeholder.Description, placeholder.CreatedAt, placeholder.LastModifiedAt);
#endif
    }

#endif
#if list
    public static PlaceholderSummaryResponse ToResponse(PlaceholderSummaryDto summary)
    {
        ArgumentNullException.ThrowIfNull(summary);

#if fields
        return new PlaceholderSummaryResponse(summary.Id, %FSummaryResponseArgs%summary.CreatedAt, summary.LastModifiedAt);
#else
        return new PlaceholderSummaryResponse(summary.Id, summary.Name, summary.CreatedAt, summary.LastModifiedAt);
#endif
    }

    public static Result<PagedResponse<PlaceholderSummaryResponse>> ToPageResponse(
        Result<PagedResult<PlaceholderSummaryDto>> result) =>
        result.Map(value => PagedResponse.From(value, ToResponse));

#endif
#if dto
#if etag
    public static Result<%ReadResponse%> ToPlaceholderResponse(Result<%ReadDto%> result) =>
        result.Map(value => new Versioned<PlaceholderResponse>(ToResponse(value.Value), value.Version));
#else
    public static Result<%ReadResponse%> ToPlaceholderResponse(Result<%ReadDto%> result) =>
        result.Map(ToResponse);
#endif
#endif
}
