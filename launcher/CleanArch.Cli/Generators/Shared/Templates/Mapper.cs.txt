using AppTemplate.Domain.Core.Common.Primitives; //?UserId
using AppTemplate.Domain.Features.Placeholders.Model.Entities;
using AppTemplate.Infrastructure.Core.Common.Saving.Tracking;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Mapping;

internal sealed class PlaceholderMapper : IPlaceholderMapper
{
    public Placeholder ToAggregate(PlaceholderRecord record)
    {
        ArgumentNullException.ThrowIfNull(record);

        var aggregate = Placeholder.Rehydrate(
            record.Id,
#if owner
            UserId.Create(record.OwnerId),
#endif
#if fields
%FMapperRehydrate%
#else
            record.Name,
            record.Description);
#endif

        StoredStamps.ApplyTo(aggregate, record, record.Version, record.Id, "Placeholder");

        return aggregate;
    }

    public PlaceholderRecord ToNewRecord(Placeholder aggregate)
    {
        ArgumentNullException.ThrowIfNull(aggregate);

        return new PlaceholderRecord
        {
            Id = aggregate.Id,
#if owner
            OwnerId = aggregate.OwnerId.Value,
#endif
#if fields
%FMapperNew%
#else
            Name = aggregate.Name,
            Description = aggregate.Description,
#endif

            Version = aggregate.Version,

            CreatedAt = aggregate.CreatedAt,
            CreatedBy = aggregate.CreatedBy,
            LastModifiedAt = aggregate.LastModifiedAt,
            LastModifiedBy = aggregate.LastModifiedBy,
        };
    }

    public void WriteTo(Placeholder aggregate, PlaceholderRecord record)
    {
        ArgumentNullException.ThrowIfNull(aggregate);
        ArgumentNullException.ThrowIfNull(record);

#if fields
%FMapperWrite%
#else
        record.Name = aggregate.Name;
        record.Description = aggregate.Description;
#endif
    }
}
