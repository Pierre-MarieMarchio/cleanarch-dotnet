using AppTemplate.Api.Common.Contracts; //?PagedResponse
using AppTemplate.Api.Common.Controllers;
using AppTemplate.Api.Common.Errors; //?ToActionResult
using AppTemplate.Api.Common.Idempotency; //?Idempotent
using AppTemplate.Api.Features.Placeholders.Contracts.Mapping; //?PlaceholderResponseMapping
using AppTemplate.Api.Features.Placeholders.Contracts.Requests; //?CreatePlaceholderRequest|UpdatePlaceholderRequest|GetPlaceholdersRequest
using AppTemplate.Api.Features.Placeholders.Contracts.Responses; //?PlaceholderResponse|PlaceholderSummaryResponse
#if record
using AppTemplate.Application.Features.Placeholders.Model.Dtos; //?PlaceholderContentDto|PlaceholderListQueryDto
using AppTemplate.Application.Features.Placeholders.Services;
#else
#if create
using AppTemplate.Application.Features.Placeholders.UseCases.Commands.CreatePlaceholder;
#endif
#if update
using AppTemplate.Application.Features.Placeholders.UseCases.Commands.UpdatePlaceholder;
#endif
#if delete
using AppTemplate.Application.Features.Placeholders.UseCases.Commands.DeletePlaceholder;
#endif
#if get
using AppTemplate.Application.Features.Placeholders.UseCases.Queries.GetPlaceholder;
#endif
#if list
using AppTemplate.Application.Features.Placeholders.UseCases.Queries.GetPlaceholders;
#endif
using Mediator;
#endif
using Microsoft.AspNetCore.Mvc;

namespace AppTemplate.Api.Features.Placeholders.Controllers;

[Route("api/v{version:apiVersion}/%R%")]
[Asp.Versioning.ApiVersion("1.0")]
[ProducesResponseType(StatusCodes.Status401Unauthorized, Type = typeof(ProblemDetails))]
public sealed class PlaceholdersController(
    %CtorParams%) : ApiControllerBase
{
#if list
    /// <summary>Lists %Scope%Placeholders, sorted and paginated%ListTail%.</summary>
    [HttpGet]
    [ProducesResponseType(StatusCodes.Status200OK, Type = typeof(PagedResponse<PlaceholderSummaryResponse>))]
    [ProducesResponseType(StatusCodes.Status400BadRequest, Type = typeof(ValidationProblemDetails))]
    public async Task<ActionResult<PagedResponse<PlaceholderSummaryResponse>>> GetAll(
        [FromQuery] GetPlaceholdersRequest request,
        CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        var query = new %ListInput%(request.Paging, request.Page, request.PageSize, request.Cursor, request.Sort);

        return OkOrProblem(PlaceholderResponseMapping.ToPageResponse(await %CallList%));
    }

#endif
#if get
#if etag
    /// <summary>Gets one placeholder, and the <c>ETag</c> needed to change it.</summary>
#else
    /// <summary>Gets one placeholder.</summary>
#endif
    [HttpGet("{placeholderId:guid}", Name = nameof(GetPlaceholder))]
    [HttpHead("{placeholderId:guid}")]
    [ProducesResponseType(StatusCodes.Status200OK, Type = typeof(PlaceholderResponse))]
#if etag
    [ProducesResponseType(StatusCodes.Status304NotModified)]
#endif
    [ProducesResponseType(StatusCodes.Status404NotFound, Type = typeof(ProblemDetails))]
    public async Task<ActionResult<PlaceholderResponse>> GetPlaceholder(Guid placeholderId, CancellationToken cancellationToken) =>
        OkOrProblem(PlaceholderResponseMapping.ToPlaceholderResponse(await %CallGet%));

#endif
#if create
    /// <summary>Creates %a% placeholder; an <c>Idempotency-Key</c> header makes a retried request safe.</summary>
    [HttpPost]
    [Idempotent]
    [ProducesResponseType(StatusCodes.Status201Created, Type = typeof(PlaceholderResponse))]
    [ProducesResponseType(StatusCodes.Status400BadRequest, Type = typeof(ValidationProblemDetails))]
    [ProducesResponseType(StatusCodes.Status409Conflict, Type = typeof(ProblemDetails))]
    public async Task<ActionResult<PlaceholderResponse>> Create(
        [FromBody] CreatePlaceholderRequest request,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(request);

        var outcome = await %CallCreate%;
        var result = PlaceholderResponseMapping.ToPlaceholderResponse(outcome);

#if etag
        return CreatedOrProblem(result, nameof(GetPlaceholder), created => new { placeholderId = created.Id });
#else
        if (result.IsFailure)
        {
            return result.Error!.ToActionResult(HttpContext);
        }

        return CreatedOrProblem(result, nameof(GetPlaceholder), new { placeholderId = result.Value.Id });
#endif
    }

#endif
#if update
#if etag
    /// <summary>Replaces %a% placeholder's fields, on the version an <c>If-Match</c> names.</summary>
#else
    /// <summary>Replaces %a% placeholder's fields; of two concurrent writers the later one wins.</summary>
#endif
    [HttpPut("{placeholderId:guid}")]
    [ProducesResponseType(StatusCodes.Status200OK, Type = typeof(PlaceholderResponse))]
    [ProducesResponseType(StatusCodes.Status400BadRequest, Type = typeof(ValidationProblemDetails))]
    [ProducesResponseType(StatusCodes.Status404NotFound, Type = typeof(ProblemDetails))]
#if etag
    [ProducesResponseType(StatusCodes.Status409Conflict, Type = typeof(ProblemDetails))]
    [ProducesResponseType(StatusCodes.Status412PreconditionFailed, Type = typeof(ProblemDetails))]
    [ProducesResponseType(StatusCodes.Status428PreconditionRequired, Type = typeof(ProblemDetails))]
#endif
    public async Task<ActionResult<PlaceholderResponse>> Update(
        Guid placeholderId,
        [FromBody] UpdatePlaceholderRequest request,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(request);

#if etag
        if (ReadPrecondition(out var precondition, out bool requiresExistence) is { } refusal)
        {
            return refusal;
        }

        var outcome = await %CallUpdate%;
        var result = RequiringExistence(requiresExistence, outcome);

        return UpdatedOrProblem(PlaceholderResponseMapping.ToPlaceholderResponse(result));
#else
        var outcome = await %CallUpdate%;

        return OkOrProblem(PlaceholderResponseMapping.ToPlaceholderResponse(outcome));
#endif
    }

#endif
#if delete
#if (update && etag)
    /// <summary>Deletes %a% placeholder, on the same <c>If-Match</c> terms as the update.</summary>
#else
    /// <summary>Deletes %a% placeholder.</summary>
#endif
    [HttpDelete("{placeholderId:guid}")]
    [ProducesResponseType(StatusCodes.Status204NoContent)]
#if etag
    [ProducesResponseType(StatusCodes.Status400BadRequest, Type = typeof(ValidationProblemDetails))]
#endif
    [ProducesResponseType(StatusCodes.Status404NotFound, Type = typeof(ProblemDetails))]
#if etag
    [ProducesResponseType(StatusCodes.Status409Conflict, Type = typeof(ProblemDetails))]
    [ProducesResponseType(StatusCodes.Status412PreconditionFailed, Type = typeof(ProblemDetails))]
    [ProducesResponseType(StatusCodes.Status428PreconditionRequired, Type = typeof(ProblemDetails))]
#endif
    public async Task<ActionResult> Delete(Guid placeholderId, CancellationToken cancellationToken)
    {
#if etag
        if (ReadPrecondition(out var precondition, out bool requiresExistence) is { } refusal)
        {
            return refusal;
        }

        var outcome = await %CallDelete%;

        return NoContentOrProblem(RequiringExistence(requiresExistence, outcome));
#else
        return NoContentOrProblem(await %CallDelete%);
#endif
    }
#endif
}
