using FluentValidation;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Queries.GetPlaceholder;

public sealed class GetPlaceholderQueryValidator : AbstractValidator<GetPlaceholderQuery>
{
    public GetPlaceholderQueryValidator() =>
        RuleFor(query => query.PlaceholderId)
            .NotEmpty().WithMessage("%An% placeholder id is required.");
}
