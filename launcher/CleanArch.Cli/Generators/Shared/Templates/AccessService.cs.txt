using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Model.Errors;
using AppTemplate.Domain.Features.Placeholders.Model.Entities;
using AppTemplate.Domain.Features.Placeholders.Repositories;

namespace AppTemplate.Application.Features.Placeholders.Services;

internal sealed class PlaceholderService(IPlaceholderRepository repository, ICurrentUser currentUser) : IPlaceholderService
{
    public async Task<Result<Placeholder>> %Load%(
        Guid placeholderId,
        VersionPrecondition? precondition,
        CancellationToken cancellationToken = default)
    {
        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<Placeholder>();
        }

        var aggregate = await repository.GetAsync(%SvcOwner%placeholderId, cancellationToken);

        if (aggregate is null)
        {
            return Result.Failure<Placeholder>(PlaceholderErrors.PlaceholderNotFound(placeholderId));
        }

        return precondition is not null && !precondition.IsSatisfiedBy(aggregate.Version)
            ? Result.Failure<Placeholder>(ConcurrencyErrors.PreconditionFailed)
            : Result.Success(aggregate);
    }
}
