using AppTemplate.Application.Core.Common.Concurrency;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Errors;
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderQueries;
using Mediator;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Queries.GetPlaceholder;

public sealed record GetPlaceholderQuery(Guid PlaceholderId) : IQuery<Result<Versioned<PlaceholderDto>>>;

public sealed class GetPlaceholderQueryHandler(
    IPlaceholderQueries queries,
    ICurrentUser currentUser) : IQueryHandler<GetPlaceholderQuery, Result<Versioned<PlaceholderDto>>>
{
    public async ValueTask<Result<Versioned<PlaceholderDto>>> Handle(
        GetPlaceholderQuery query,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(query);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<Versioned<PlaceholderDto>>();
        }

        var found = await queries.GetAsync(%SvcOwner%query.PlaceholderId, cancellationToken);

        return found is null
            ? Result.Failure<Versioned<PlaceholderDto>>(PlaceholderErrors.PlaceholderNotFound(query.PlaceholderId))
            : found;
    }
}
