#if rec
using AppTemplate.Application.Features.Placeholders.Model.Policies; //?PlaceholderInputPolicy
#else
using AppTemplate.Domain.Features.Placeholders.Model.Entities; //?Placeholder
#endif
using AppTemplate.Infrastructure.Core.Common.Contexts; //?IsVersion
using AppTemplate.Infrastructure.Persistence.Common.Contexts;
using AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Models;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace AppTemplate.Infrastructure.Persistence.Features.Placeholders.Rows.Configurations;

internal sealed class PlaceholderRecordConfiguration : IEntityTypeConfiguration<PlaceholderRecord>
{
    public void Configure(EntityTypeBuilder<PlaceholderRecord> builder)
    {
        ArgumentNullException.ThrowIfNull(builder);

        builder.ToTable("Placeholders", AppDbContext.PlaceholdersSchema);

        builder.HasKey(record => record.Id);

        builder.Property(record => record.Id).ValueGeneratedNever();

#if owner
        builder.Property(record => record.OwnerId).IsRequired();

#endif
#if list
#if owner
#if fields
%FIndexes%
#else
        builder.HasIndex(record => new { record.OwnerId, record.Name, record.Id })
            .HasDatabaseName("IX_Placeholders_OwnerId_Name_Id");
#endif

        builder.HasIndex(record => new { record.OwnerId, record.CreatedAt, record.Id })
            .HasDatabaseName("IX_Placeholders_OwnerId_CreatedAt_Id");
#else
#if fields
%FIndexes%
#else
        builder.HasIndex(record => new { record.Name, record.Id })
            .HasDatabaseName("IX_Placeholders_Name_Id");
#endif

        builder.HasIndex(record => new { record.CreatedAt, record.Id })
            .HasDatabaseName("IX_Placeholders_CreatedAt_Id");
#endif

#else
#if owner
        builder.HasIndex(record => record.OwnerId)
            .HasDatabaseName("IX_Placeholders_OwnerId");

#endif
#endif
#if fields
%FConfiguration%
#else
        builder.Property(record => record.Name)
            .HasMaxLength(%Limits%.NameMaxLength)
            .IsRequired();

        builder.Property(record => record.Description)
            .HasMaxLength(%Limits%.DescriptionMaxLength)
            .IsRequired();
#endif

        builder.Property(record => record.CreatedAt).IsRequired();
#if agg
        builder.Property(record => record.CreatedBy);
#endif
        builder.Property(record => record.LastModifiedAt);
#if agg
        builder.Property(record => record.LastModifiedBy);
#endif
#if etag

        builder.Property(record => record.Version).IsVersion();
#endif
    }
}
