using AppTemplate.Application.Core.Common.Collections;
using AppTemplate.Application.Core.Common.Ports;
using AppTemplate.Application.Core.Common.Results;
using AppTemplate.Application.Features.Placeholders.Model.Dtos;
using AppTemplate.Application.Features.Placeholders.Model.Policies;
using AppTemplate.Application.Features.Placeholders.Ports.PlaceholderQueries;
using Mediator;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Queries.GetPlaceholders;

public sealed record GetPlaceholdersQuery(
    string? Paging,
    int? Page,
    int? PageSize,
    string? Cursor,
    string? Sort) : ICollectionQuery, IQuery<Result<PagedResult<PlaceholderSummaryDto>>>
{
    public static GetPlaceholdersQuery Offset(int? page, int? pageSize) => new(null, page, pageSize, null, null);
}

public sealed class GetPlaceholdersQueryHandler(IPlaceholderQueries queries, ICurrentUser currentUser)
    : IQueryHandler<GetPlaceholdersQuery, Result<PagedResult<PlaceholderSummaryDto>>>
{
    public async ValueTask<Result<PagedResult<PlaceholderSummaryDto>>> Handle(
        GetPlaceholdersQuery query,
        CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(query);

        var userId = currentUser.RequireUserId();

        if (userId.IsFailure)
        {
            return userId.To<PagedResult<PlaceholderSummaryDto>>();
        }

        var bound = CollectionBinding.Bind(
            query,
            PlaceholderCollectionPolicy.Instance,
            () => Result.Success(PlaceholderFilter.None),
            %DateKeys%);

        if (bound.IsFailure)
        {
            return bound.To<PagedResult<PlaceholderSummaryDto>>();
        }

        return await queries.%ListQuery%(%SvcOwner%bound.Value, cancellationToken);
    }
}
