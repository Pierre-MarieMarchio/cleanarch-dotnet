using FluentValidation;

namespace AppTemplate.Application.Features.Placeholders.UseCases.Commands.DeletePlaceholder;

public sealed class DeletePlaceholderCommandValidator : AbstractValidator<DeletePlaceholderCommand>
{
    public DeletePlaceholderCommandValidator() =>
        RuleFor(command => command.PlaceholderId)
            .NotEmpty().WithMessage("%An% placeholder id is required.");
}
