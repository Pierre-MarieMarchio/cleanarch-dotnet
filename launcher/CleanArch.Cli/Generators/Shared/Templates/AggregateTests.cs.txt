using AppTemplate.Domain.Core.Common.Exceptions;
using AppTemplate.Domain.Core.Common.Primitives; //?UserId
using AppTemplate.Domain.Features.Placeholders.Model.Entities;
using Shouldly;
using Xunit;

namespace AppTemplate.Domain.UnitTests.Features.Placeholders.Model.Entities;

public sealed class PlaceholderTests
{
#if owner
    private static readonly UserId _ownerId = UserId.Create(Guid.CreateVersion7());

#endif
#if fields
%FAggregateTests%
#else
#if create
    [Fact]
    public void Create_KeepsTheTrimmedNameAndTheDescription()
    {
        var placeholder = Placeholder.Create(%OwnerIdArg%"  Groceries  ", "Milk, eggs");

        placeholder.Id.ShouldNotBe(Guid.Empty);
        placeholder.Name.ShouldBe("Groceries");
        placeholder.Description.ShouldBe("Milk, eggs");
#if owner
        placeholder.OwnerId.ShouldBe(_ownerId);
#endif
    }

    [Theory]
    [InlineData("")]
    [InlineData("   ")]
    public void Create_RefusesABlankName(string name) =>
        Should.Throw<DomainException>(() => Placeholder.Create(%OwnerIdArg%name, "Milk, eggs"));

    [Fact]
    public void Create_RefusesANameOverTheLimit() =>
        Should.Throw<DomainException>(
            () => Placeholder.Create(%OwnerIdArg%new string('a', Placeholder.NameMaxLength + 1), string.Empty));

#endif
#if update
    [Fact]
    public void Update_ReplacesTheNameAndTheDescription()
    {
        var placeholder = Placeholder.Rehydrate(Guid.CreateVersion7(), %OwnerIdArg%"Groceries", "Milk");

        placeholder.Update("  Errands  ", "Parcel");

        placeholder.Name.ShouldBe("Errands");
        placeholder.Description.ShouldBe("Parcel");
    }

    [Fact]
    public void Update_RefusesABlankName()
    {
        var placeholder = Placeholder.Rehydrate(Guid.CreateVersion7(), %OwnerIdArg%"Groceries", "Milk");

        Should.Throw<DomainException>(() => placeholder.Update("   ", "Parcel"));
        placeholder.Name.ShouldBe("Groceries");
    }

    [Fact]
    public void Update_RefusesAMissingDescription_WithoutChangingTheName()
    {
        var placeholder = Placeholder.Rehydrate(Guid.CreateVersion7(), %OwnerIdArg%"Groceries", "Milk");

        Should.Throw<DomainException>(() => placeholder.Update("Errands", null!));
        placeholder.Name.ShouldBe("Groceries");
        placeholder.Description.ShouldBe("Milk");
    }

#endif
    [Fact]
    public void Rehydrate_KeepsWhatWasStored_EvenOverTheCurrentLimit()
    {
        string stored = new('a', Placeholder.NameMaxLength + 1);

        var placeholder = Placeholder.Rehydrate(Guid.CreateVersion7(), %OwnerIdArg%stored, "Milk");

        placeholder.Name.ShouldBe(stored);
    }

    [Fact]
    public void Rehydrate_RefusesAnEmptyId() =>
        Should.Throw<DomainException>(() => Placeholder.Rehydrate(Guid.Empty, %OwnerIdArg%"Groceries", "Milk"));
#endif
}
